    // Load news updates from API
    function loadNewsUpdates() {  ← ✅ FUNCTION DEFINED
        fetch('http://localhost:5001/api/news/all')
            .then(res => res.json())
            .then(data => {
                if (data.news && data.news.length > 0) {
                    const newsHTML = data.news.slice(0, 3).map(article => `
                        <div style="background: white; border-radius: 12px; padding: 20px; box-shadow: 0 4px 15px rgba(0,0,0,0.08);">
                            <div style="height: 150px; background: linear-gradient(135deg, #FF6B35, #F7931E); border-radius: 8px; margin-bottom: 15px;"></div>
                            <h4 style="font-weight: 700; color: var(--dark); margin-bottom: 8px;">${article.title}</h4>
                            <p style="font-size: 13px; color: #666; margin-bottom: 10px;">${article.description}</p>
                            <p style="font-size: 12px; color: #999; margin-bottom: 8px;"><strong>Source:</strong> ${article.source}</p>
                            <p style="font-size: 12px; color: #999;">${new Date(article.published_at).toLocaleDateString()}</p>
                        </div>
                    `).join('');
                    
                    const container = document.getElementById('newsContainer');
                    if (container) {
                        container.innerHTML = newsHTML;  ← ✅ DISPLAYS NEWS
                    }
                }
            })
            .catch(err => console.log('News loading (using sample data)'));
    }
